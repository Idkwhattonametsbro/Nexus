/* ---------- Utility ---------- */
const STORAGE_KEY = 'habit-tracker-data';
const THEME_KEY   = 'habit-tracker-theme';
let data = { habits: [] };

function saveData(){ localStorage.setItem(STORAGE_KEY, JSON.stringify(data)); }
function loadData(){
  const raw = localStorage.getItem(STORAGE_KEY);
  if(raw){ data = JSON.parse(raw); }
}
function setTheme(theme){
  document.documentElement.setAttribute('data-theme',theme);
  localStorage.setItem(THEME_KEY,theme);
}
function toggleTheme(){
  const cur = document.documentElement.getAttribute('data-theme')||'dark';
  setTheme(cur==='dark'?'light':'dark');
}

/* ---------- Dates ---------- */
function startOfWeek(date){
  const d = new Date(date);
  const day = d.getDay(); // 0 Sun … 6 Sat
  const diff = (day===0? -6 : 1 - day); // Monday as first day
  d.setDate(d.getDate()+diff);
  d.setHours(0,0,0,0);
  return d;
}
function formatDate(d){
  return d.toISOString().split('T')[0];
}
function getWeekDays(){
  const start = startOfWeek(new Date());
  return [...Array(7)].map((_,i)=>{
    const d = new Date(start);
    d.setDate(start.getDate()+i);
    return d;
  });
}

/* ---------- Rendering ---------- */
function renderHabits(){
  const container = document.getElementById('habitsContainer');
  container.innerHTML='';
  const weekDays = getWeekDays();

  data.habits.forEach(habit=>{
    const div = document.createElement('div');
    div.className='habit';
    div.dataset.id=habit.id;

    const name = document.createElement('div');
    name.className='habit-name';
    name.textContent=habit.name;
    div.appendChild(name);

    const daysDiv = document.createElement('div');
    daysDiv.className='days';
    weekDays.forEach(day=>{
      const dayKey = formatDate(day);
      const chk = document.createElement('input');
      chk.type='checkbox';
      chk.checked = !!habit.days[dayKey];
      chk.dataset.habit = habit.id;
      chk.dataset.date  = dayKey;
      chk.addEventListener('change', onCheckChange);
      daysDiv.appendChild(chk);
    });
    div.appendChild(daysDiv);

    // day labels
    const labels = document.createElement('div');
    labels.className='days';
    weekDays.forEach(day=>{
      const lbl = document.createElement('div');
      lbl.className='day-label';
      lbl.textContent = day.toLocaleDateString(undefined,{weekday:'short'});
      labels.appendChild(lbl);
    });
    div.appendChild(labels);

    // progress bar
    const completed = weekDays.filter(d=>habit.days[formatDate(d)]).length;
    const percent = Math.round(completed/7*100);
    const prog = document.createElement('div');
    prog.className='progress';
    const inner = document.createElement('div');
    inner.style.width = percent+'%';
    prog.appendChild(inner);
    div.appendChild(prog);

    container.appendChild(div);
  });

  updateStreak();
}
function onCheckChange(e){
  const habitId = e.target.dataset.habit;
  const date    = e.target.dataset.date;
  const habit = data.habits.find(h=>h.id===habitId);
  if(!habit) return;
  if(e.target.checked){
    habit.days[date]=true;
  }else{
    delete habit.days[date];
  }
  saveData();
  renderHabits();
}
function addHabit(name){
  const id = crypto.randomUUID();
  data.habits.push({id, name, days:{}});
  saveData();
  renderHabits();
}
function updateStreak(){
  const today = new Date();
  let streak = 0;
  while(true){
    const dayKey = formatDate(today);
    const allDone = data.habits.every(h=>h.days[dayKey]);
    if(!allDone) break;
    streak++;
    today.setDate(today.getDate()-1);
  }
  document.getElementById('streakCount').textContent = streak;
}

/* ---------- Event Listeners ---------- */
document.getElementById('addForm').addEventListener('submit',e=>{
  e.preventDefault();
  const input = document.getElementById('habitInput');
  const name = input.value.trim();
  if(name){
    addHabit(name);
    input.value='';
  }
});
document.getElementById('themeToggle').addEventListener('click',toggleTheme);

/* ---------- Init ---------- */
loadData();
const savedTheme = localStorage.getItem(THEME_KEY)||'dark';
setTheme(savedTheme);
renderHabits();
</script>
