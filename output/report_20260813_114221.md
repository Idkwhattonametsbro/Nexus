<script>
(() => {
  const WORK_SEC   = 25 * 60;   // 25 minutes
  const BREAK_SEC  = 5  * 60;   // 5 minutes

  let remaining   = WORK_SEC;
  let timerId     = null;
  let isWork      = true;
  let sessions    = 0;

  const timerEl    = document.getElementById('timer');
  const startBtn   = document.getElementById('startBtn');
  const pauseBtn   = document.getElementById('pauseBtn');
  const resetBtn   = document.getElementById('resetBtn');
  const countEl    = document.getElementById('count');
  const progressEl = document.getElementById('progress');

  const fmt = s => {
    const m = String(Math.floor(s/60)).padStart(2,'0');
    const sec = String(s%60).padStart(2,'0');
    return `${m}:${sec}`;
  };

  const update = () => {
    timerEl.textContent = fmt(remaining);
    const total = isWork ? WORK_SEC : BREAK_SEC;
    const pct = ((total-remaining)/total)*100;
    progressEl.style.width = pct + '%';
  };

  const tick = () => {
    if (remaining > 0) {
      remaining--;
      update();
    } else {
      clearInterval(timerId);
      timerId = null;
      if (isWork) {
        sessions++;
        countEl.textContent = sessions;
        alert('Work session finished! Time for a short break.');
        isWork = false;
        remaining = BREAK_SEC;
      } else {
        alert('Break over! Back to work.');
        isWork = true;
        remaining = WORK_SEC;
      }
      update();
      startBtn.disabled = false;
      pauseBtn.disabled = true;
      resetBtn.disabled = true;
    }
  };

  startBtn.onclick = () => {
    if (!timerId) {
      timerId = setInterval(tick, 1000);
      startBtn.disabled = true;
      pauseBtn.disabled = false;
      resetBtn.disabled = false;
    }
  };

  pauseBtn.onclick = () => {
    if (timerId) {
      clearInterval(timerId);
      timerId = null;
      startBtn.disabled = false;
      pauseBtn.disabled = true;
    }
  };

  resetBtn.onclick = () => {
    clearInterval(timerId);
    timerId = null;
    remaining = isWork ? WORK_SEC : BREAK_SEC;
    update();
    startBtn.disabled = false;
    pauseBtn.disabled = true;
    resetBtn.disabled = true;
  };

  // initial render
  update();
})();
</script>
</body>
