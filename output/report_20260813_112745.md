<script>
// Simulate live KPI updates (demo only)
function randomInt(min, max) { return Math.floor(Math.random() * (max - min + 1)) + min; }
setInterval(() => {
  document.getElementById('cpu').textContent = randomInt(20, 80) + '%';
  document.getElementById('mem').textContent = randomInt(40, 90) + '%';
  document.getElementById('tasks').textContent = randomInt(100, 150);
  document.getElementById('errors').textContent = randomInt(0, 5);
}, 5000);

// Log handling
const logContainer = document.getElementById('execLog');
const logInput = document.getElementById('logInput');
const addBtn = document.getElementById('addLogBtn');

addBtn.addEventListener('click', () => {
  const text = logInput.value.trim();
  if (!text) return;
  const entry = document.createElement('div');
  entry.className = 'log-entry';
  const now = new Date().toLocaleTimeString();
  entry.innerHTML = `<time>${now}</time>${text}`;
  logContainer.appendChild(entry);
  logContainer.scrollTop = logContainer.scrollHeight;
  logInput.value = '';
});

logInput.addEventListener('keypress', e => {
  if (e.key === 'Enter') addBtn.click();
});
</script>
